\NotatioSetArabicFont{Amiri} % Example font; not bundled.
\begin{document}
\section*{Arabic notation: interface demonstration}
This sample demonstrates Arabic-script symbols and Arabic-Indic digits
inside ordinary left-to-right mathematics. It does not implement
right-to-left formula layout.
\[
  \NotatioArabicMath{س} + \NotatioArabicNumeral{12.5}
  = \NotatioArabicNumeral{13.5}
\]
